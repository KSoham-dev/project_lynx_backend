"""
pipeline/agents/image_identifier.py
======================================
Layer 1 — ImageIdentifier agent

Two-stage species identification:

  Stage 1 — SpeciesNet classification (high confidence):
    • Extract top classification label + score from the prediction.
    • Parse the scientific name from SpeciesNet's semicolon-delimited taxonomy label:
        "animalia;chordata;mammalia;carnivora;felidae;panthera;leo"
        → "Panthera leo"
    • If score >= CONFIDENCE_THRESHOLD (0.80) → return scientific name, source="model".

  Stage 2 — Azure OpenAI GPT vision fallback (low confidence):
    • Called when model score < 0.80 or when label parsing fails.
    • Sends the original image URL to the GptPlugin with a zoologist system prompt.
    • Parses the JSON response for "species_name" field.
    • Returns scientific name, source="gpt".
"""
from __future__ import annotations

import json
import logging

from semantic_kernel.functions import kernel_function

from .gpt_function import GptPlugin, SPECIES_ID_SYSTEM_PROMPT
from .schemas import IdentificationResult

logger = logging.getLogger(__name__)

# ── Configuration ──────────────────────────────────────────────────────────────
CONFIDENCE_THRESHOLD = 0.80  # SpeciesNet score required to trust model label

# Tokens that appear in taxonomy labels but are not valid species names
_NON_SPECIES_TOKENS = frozenset(
    {"blank", "vehicle", "animal", "human", "person", "other", "unknown", ""}
)


def _parse_scientific_name(label: str) -> str | None:
    """
    Extract the scientific name from a SpeciesNet taxonomy label.

    SpeciesNet format (semicolon-separated from kingdom → species):
        "animalia;chordata;mammalia;carnivora;felidae;panthera;leo"
        → genus="Panthera", epithet="leo"  →  "Panthera leo"

    Returns None when:
      - Fewer than 2 taxonomy levels are present.
      - The species epithet is a known non-species token (blank / vehicle / …).
    """
    parts = [p.strip() for p in label.split(";") if p.strip()]
    if len(parts) < 2:
        return None

    genus, epithet = parts[-2].lower(), parts[-1].lower()

    if epithet in _NON_SPECIES_TOKENS or genus in _NON_SPECIES_TOKENS:
        return None

    return f"{genus.capitalize()} {epithet}"


class ImageIdentifierPlugin:
    """
    Layer 1 — identifies species from a SpeciesNet prediction.

    Requires a GptPlugin instance injected at construction time so that
    the GPT fallback shares the same client configuration.

    Plugin name registered in Kernel: "identification"
    """

    def __init__(self, gpt_plugin: GptPlugin) -> None:
        self._gpt = gpt_plugin

    @kernel_function(
        name="identify",
        description=(
            "Identify the species in an image using SpeciesNet classifications "
            "or GPT-4o-mini vision as a fallback."
        ),
    )
    def identify(self, prediction: dict, image_url: str) -> IdentificationResult:
        """
        Identify species from a SpeciesNet prediction dict.

        Args:
            prediction: Single prediction entry from SpeciesNet output.
            image_url:  Original image URL — passed to GPT if fallback is needed.

        Returns:
            IdentificationResult with scientific_name, confidence, and source.
        """
        classifications: dict = prediction.get("classifications") or {}
        classes: list[str] = classifications.get("classes") or []
        scores:  list[float] = classifications.get("scores") or []

        if classes and scores:
            top_label = classes[0]
            top_score = float(scores[0])
            scientific_name = _parse_scientific_name(top_label)

            if scientific_name and top_score >= CONFIDENCE_THRESHOLD:
                logger.info(
                    "Model identified %r with confidence %.2f", scientific_name, top_score
                )
                return IdentificationResult(
                    scientific_name=scientific_name,
                    confidence=top_score,
                    source="model",
                )

            logger.info(
                "Model confidence %.2f < %.2f threshold%s — falling back to GPT vision.",
                top_score,
                CONFIDENCE_THRESHOLD,
                f" (label={top_label!r})" if not scientific_name else "",
            )
        else:
            logger.info("No classifications in prediction — falling back to GPT vision.")

        # ── GPT vision fallback ───────────────────────────────────────────────
        return self._identify_via_gpt(image_url)

    # ── Private helpers ───────────────────────────────────────────────────────

    def _identify_via_gpt(self, image_url: str) -> IdentificationResult:
        """Call GPT vision and parse its JSON response for the scientific name."""
        try:
            raw = self._gpt.call_gpt(
                user_prompt="What species is in this image? Return only the JSON.",
                system_prompt=SPECIES_ID_SYSTEM_PROMPT,
                image_url=image_url,
                json_response=True,
            )
            data: dict = json.loads(raw)
            # Accept either "species_name" or "scientific_name" from GPT output
            species_name: str = (
                data.get("species_name")
                or data.get("scientific_name")
                or "Unknown"
            )
            logger.info("GPT identified species as %r", species_name)
            return IdentificationResult(
                scientific_name=species_name,
                confidence=0.0,  # GPT does not return a numeric confidence
                source="gpt",
            )
        except Exception as exc:
            logger.error("GPT vision fallback failed: %s", exc)
            return IdentificationResult(
                scientific_name="Unknown",
                confidence=0.0,
                source="gpt_error",
            )
